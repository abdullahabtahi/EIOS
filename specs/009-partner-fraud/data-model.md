## 1. Domain Entities & Schemas

### 1.1 `TradingPairCorrelation`
Represents an identified coordinated trading relationship between two accounts referred by distinct affiliates.

```python
class DirectionalType(StrEnum):
    OPPOSITE = "OPPOSITE"      # Account A buys while Account B sells
    MIRROR = "MIRROR"          # Accounts trade in identical direction and sizing
    UNCORRELATED = "UNCORRELATED"

class TradingPairCorrelation(BaseModel):
    pair_id: str = Field(..., description="Unique deterministic pair identifier")
    account_a: str = Field(..., description="First client account identifier")
    partner_a: str = Field(..., description="Affiliate ID referring account A")
    account_b: str = Field(..., description="Second client account identifier")
    partner_b: str = Field(..., description="Affiliate ID referring account B")
    symbol: str = Field(..., description="Financial instrument traded (e.g. EURUSD)")
    co_occurrence_count: int = Field(..., ge=1, description="Number of correlated trade events observed")
    timing_correlation_pct: float = Field(..., ge=0.0, le=100.0, description="Percentage of trades executed within 15s window")
    directional_type: DirectionalType
    size_symmetry_score: float = Field(..., ge=0.0, le=1.0, description="Volume symmetry between the two accounts")
    combined_net_pl_usd: float = Field(..., description="Combined net trading profit/loss across the pair")
    total_commission_extracted_usd: float = Field(..., ge=0.0, description="Total partner commissions generated by the pair")
    zero_sum_ratio: float = Field(..., ge=0.0, le=1.0, description="Measure of zero-sum volume extraction")
    collusion_confidence: float = Field(..., ge=0.0, le=1.0, description="Composite collusion probability")
```

class ReferralRingNode(BaseModel):

### 1.6 `PartnerFraudCase`
The comprehensive investigation dossier entity.

```python
class FraudTypology(StrEnum):
    OPPOSITE_TRADING_SYNDICATE = "OPPOSITE_TRADING_SYNDICATE"
    MIRROR_TRADING_RING = "MIRROR_TRADING_RING"
    BONUS_CPA_HARVESTING = "BONUS_CPA_HARVESTING"
    MULTI_TIER_COMMISSION_FRAUD = "MULTI_TIER_COMMISSION_FRAUD"
    CLIENT_RECYCLING = "CLIENT_RECYCLING"
    BOT_TRAFFIC_FUNNEL = "BOT_TRAFFIC_FUNNEL"

class PartnerFraudCase(BaseModel):
    case_id: str = Field(..., description="Unique case reference CASE-FRAUD-<UUID>")
    partner_id: str
    master_partner_id: str | None = None
    primary_typology: FraudTypology
    fraud_risk_score: float = Field(..., ge=0.0, le=1.0)
    total_exposure_usd: float = Field(..., ge=0.0)
    confidence_level: Literal["LOW", "MEDIUM", "HIGH", "CRITICAL"]
    implicated_accounts: list[str]
    evidence_refs: list[str]
    narrative_summary: str
    recommended_action: Literal["ClawbackCommission", "SuspendPartner", "AdjustPartnerTier", "ReviewAudit"]
    created_at: datetime = Field(default_factory=lambda: datetime.now(UTC))
    status: Literal["OPEN", "UNDER_REVIEW", "ACTIONED", "DISMISSED"]
```

## 2. DuckDB Hot-State Database Schemas

These analytical tables are initialized in `HotStateEngine` to support real-time OMCP query responses ($< 50\text{ms}$).

```sql
CREATE TABLE IF NOT EXISTS partner_fraud_scores (
    partner_id VARCHAR PRIMARY KEY,
    master_partner_id VARCHAR,
    tier_level INTEGER NOT NULL,
    primary_typology VARCHAR NOT NULL,
    fraud_risk_score DOUBLE NOT NULL,
    client_quality_index DOUBLE NOT NULL,
    opposite_trading_pair_count INTEGER NOT NULL,
    bonus_abuse_probability DOUBLE NOT NULL,
    bot_traffic_probability DOUBLE NOT NULL,
    total_exposure_usd DOUBLE NOT NULL,
    computed_at TIMESTAMP NOT NULL
);

CREATE TABLE IF NOT EXISTS partner_opposite_trading_pairs (
    pair_id VARCHAR PRIMARY KEY,
    partner_a VARCHAR NOT NULL,
    account_a VARCHAR NOT NULL,
    partner_b VARCHAR NOT NULL,
    account_b VARCHAR NOT NULL,
    symbol VARCHAR NOT NULL,
    co_occurrence_count INTEGER NOT NULL,
    timing_correlation_pct DOUBLE NOT NULL,
    directional_type VARCHAR NOT NULL,
    combined_net_pl_usd DOUBLE NOT NULL,
    total_commission_usd DOUBLE NOT NULL,
    collusion_confidence DOUBLE NOT NULL,
    detected_at TIMESTAMP NOT NULL
);

CREATE TABLE IF NOT EXISTS partner_fraud_cases (
    case_id VARCHAR PRIMARY KEY,
    partner_id VARCHAR NOT NULL,
    master_partner_id VARCHAR,
    primary_typology VARCHAR NOT NULL,
    fraud_risk_score DOUBLE NOT NULL,
    total_exposure_usd DOUBLE NOT NULL,
    confidence_level VARCHAR NOT NULL,
    status VARCHAR DEFAULT 'OPEN',
    created_at TIMESTAMP NOT NULL
);
```
